% Remplace les \textbf employés comme sous-titres : couleur + gras + espace après,
% pour qu'ils se distinguent clairement du texte courant (sans numérotation ni entrée TdM).
\makeatletter
\newcommand{\paratitle}[1]{%
  \par\addvspace{2.5ex}%
  \noindent{\normalfont\normalsize\bfseries\color{bleuuu}#1}%
  \par\nobreak\vskip 1.2ex%
  \@afterheading}
